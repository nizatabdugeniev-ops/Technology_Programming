# Введите длины трех сторон треугольника. Найдите его полупериметр и площадь по формуле Герона. Предполагается, что корректный треугольник уже задан

import math

a = float(input())
b = float(input())
c = float(input())

p = (a + b + c) / 2
S = math.sqrt(p * (p - a) * (p - b) * (p - c))

print("Полупериметр:", p)
print("Площадь:", S)
